<script>
    // --- Navbar and Theme Logic ---
    document.addEventListener('DOMContentLoaded', () => {

            document.getElementById('profileDropdown').classList.toggle('hidden');
        });
        // This ensures the page theme matches the rest of the site
        if (localStorage.getItem('theme') === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
            document.documentElement.classList.add('dark');
        } else {
            document.documentElement.classList.remove('dark');
        }
    });

    // --- NEW: Refactored Curriculum Page Logic with XP Tracking ---
    const CurriculumManager = {
        nodes: {},
        state: {
            isDragging: false,
            curriculumData: {},
            userProgress: {},
            goalId: null,
            orderedStepIds: [],
            pathKeys: ["Short Duration Path", "Moderate Duration Path", "Long Duration Path"],
            xpTotals: {
                overall: 0,
                paths: {},
                phases: {}
            },
            currentPathKey: "Short Duration Path"
        },

        // HELPER FUNCTION
        buildStepMap() {
            this.state.stepMap = {};
            this.state.pathKeys.forEach(pathKey => {
                const pathData = this.state.curriculumData[pathKey] || [];
                pathData.forEach(phase => {
                    phase.Steps.forEach(step => {
                        const stepId = step.Step.replace(/[^a-zA-Z0-9]/g, '-');
                        // Store the entire step object, keyed by its ID
                        if (!this.state.stepMap[stepId]) {
                            this.state.stepMap[stepId] = step;
                        }
                    });
                });
            });
        },

        init() {
            this.cacheNodes();
            if (this.validateNodes()) {
                // 1. Get data from the template
                this.state.curriculumData = {{ curriculum_data | tojson }};
                this.state.goalId = {{ goal_id }};

                if (this.state.curriculumData && this.state.curriculumData.Title) {
                    // 2. Set up initial state and calculations
                    this.nodes.curriculumTitle.textContent = this.state.curriculumData.Title;
                    this.buildStepMap(); // Build the fast-lookup map for steps
                    this.calculateXpTotals(); // Calculate all possible XP
                    this.bindEvents(); // Bind all click/drag listeners

                    // 3. Fetch user progress and do the first render
                    this.fetchAndRenderInitialView();
                } else {
                    this.nodes.curriculumTitle.textContent = "Could not load curriculum.";
                }
            }
        },

        cacheNodes() {
            this.nodes.curriculumTitle = document.getElementById('curriculumTitle');
            this.nodes.curriculumRoadmap = document.getElementById('curriculumRoadmap');
            this.nodes.sliderThumb = document.getElementById('sliderThumb');
            this.nodes.sliderContainer = this.nodes.sliderThumb ? this.nodes.sliderThumb.parentElement : null;
            this.nodes.labels = {
                short: document.getElementById('label-short'),
                moderate: document.getElementById('label-moderate'),
                long: document.getElementById('label-long'),
            };
            // XP Section Nodes
            this.nodes.totalXpLabel = document.getElementById('total-xp-label');
            this.nodes.totalXpBar = document.getElementById('total-xp-bar');
            this.nodes.redeemableXpLabel = document.getElementById('redeemable-xp-label');
            this.nodes.xpDurationContainer = document.getElementById('xp-duration-container');
        },

        // CORRECTED validateNodes() function
        validateNodes() {
            const essentialNodes = [
                this.nodes.curriculumTitle,
                this.nodes.curriculumRoadmap,
                this.nodes.sliderThumb,
                this.nodes.sliderContainer,
                ...Object.values(this.nodes.labels),
                // **FIX**: Add the new, essential XP nodes to the validation check
                this.nodes.totalXpLabel,
                this.nodes.totalXpBar,
                this.nodes.redeemableXpLabel,
                this.nodes.xpDurationContainer
            ];
            
            if (essentialNodes.some(node => !node)) {
                console.error("Curriculum Manager Error: One or more essential HTML elements could not be found. Please check your element IDs. Halting execution.");
                // Try to set an error message on the title if it exists, otherwise log it.
                if (this.nodes.curriculumTitle) {
                    this.nodes.curriculumTitle.textContent = "Error: Page elements not found.";
                }
                return false;
            }
            return true;
        },

        // --- 3. Event Binding (No changes here, but included for completeness) ---
        bindEvents() {
            // Slider Events
            this.nodes.sliderThumb.addEventListener('mousedown', () => { this.state.isDragging = true; });
            document.addEventListener('mouseup', () => {
                if (this.state.isDragging) {
                    this.state.isDragging = false;
                    this.snapSliderAndRender();
                }
            });
            document.addEventListener('mousemove', (e) => {
                if (!this.state.isDragging) return;
                const rect = this.nodes.sliderContainer.getBoundingClientRect();
                const newX = e.clientX - rect.left;
                const percent = (newX / rect.width) * 100;
                this.updateSliderUI(percent);
            });
            Object.entries(this.nodes.labels).forEach(([key, label], index) => {
                label.addEventListener('click', () => {
                    const snapPos = index * 50;
                    this.updateSliderUI(snapPos, true);
                    this.renderCurriculum(this.state.pathKeys[index]);
                });
            });

            // Event Delegation for dynamic content in the roadmap
            this.nodes.curriculumRoadmap.addEventListener('click', this.handleRoadmapClick.bind(this));
            this.nodes.curriculumRoadmap.addEventListener('change', this.handleFileInputChange.bind(this));
            this.nodes.curriculumRoadmap.addEventListener('dragover', this.handleDragOver.bind(this));
            this.nodes.curriculumRoadmap.addEventListener('dragleave', this.handleDragLeave.bind(this));
            this.nodes.curriculumRoadmap.addEventListener('drop', this.handleDrop.bind(this));
        },

        // XP Calculator
        calculateXpTotals() {
            let overallTotal = 0;
            this.state.pathKeys.forEach(pathKey => {
                const pathData = this.state.curriculumData[pathKey] || [];
                let pathTotal = 0;

                // **FIX**: Initialize the objects to store the totals
                this.state.xpTotals.phases[pathKey] = {};

                pathData.forEach((phase, phaseIndex) => {
                    let phaseTotal = 0;
                    phase.Steps.forEach(step => {
                        phaseTotal += step.XPs || 0;
                    });

                    // **FIX**: Store the total for this specific phase
                    this.state.xpTotals.phases[pathKey][`phase-${phaseIndex}`] = phaseTotal;
                    pathTotal += phaseTotal;
                });

                // **FIX**: Store the total for this specific path
                this.state.xpTotals.paths[pathKey] = pathTotal;
                overallTotal += pathTotal;
            });
            this.state.xpTotals.overall = overallTotal;
        },

        // Runs every time we switch duration paths or upload a proof
        renderXpSection() {
            const { curriculumData, userProgress, xpTotals, currentPathKey } = this.state;
            
            // 1. Calculate Overall Progress
            let totalEarnedXp = 0;
            Object.keys(userProgress).forEach(stepId => {
                const step = this.findStepById(stepId);
                if (step) totalEarnedXp += step.XPs;
            });
            const overallPercentage = xpTotals.overall > 0 ? (totalEarnedXp / xpTotals.overall) * 100 : 0;
            this.nodes.totalXpBar.style.width = `${overallPercentage}%`;
            this.nodes.totalXpLabel.textContent = `${totalEarnedXp} / ${xpTotals.overall} XP`;
            this.nodes.redeemableXpLabel.textContent = `${xpTotals.overall - totalEarnedXp}`;

            // 2. Show the correct duration section
            document.querySelectorAll('.duration-xp-section').forEach(el => el.classList.add('hidden'));
            const activeXpSection = document.getElementById(`xp-${currentPathKey.split(' ')[0].toLowerCase()}-path`);
            if (activeXpSection) {
                activeXpSection.classList.remove('hidden');
                activeXpSection.innerHTML = ''; // Clear previous content
            }

            // 3. Render progress for the active path
            const pathData = curriculumData[currentPathKey] || [];
            let pathEarnedXp = 0;
            let phaseUnlocked = true;

            pathData.forEach((phase, phaseIndex) => {
                if (!phaseUnlocked) return;

                let phaseEarnedXp = 0;
                let allStepsInPhaseComplete = true;
                
                const stepsHtml = phase.Steps.map(step => {
                    const stepId = step.Step.replace(/[^a-zA-Z0-9]/g, '-');
                    const isComplete = !!userProgress[stepId];
                    if (isComplete) {
                        phaseEarnedXp += step.XPs;
                    } else {
                        allStepsInPhaseComplete = false;
                    }
                    const stepPercentage = isComplete ? 100 : 0;
                    return `
                        <div class="ml-4">
                            <div class="flex justify-between text-xs mb-1"><span class="font-medium">${step.Step}</span><span>${isComplete ? step.XPs : 0} / ${step.XPs} XP</span></div>
                            <div class="w-full bg-slate-200 dark:bg-slate-700 rounded-full h-1.5"><div class="bg-gradient-to-r from-blue-400 to-cyan-300 h-1.5 rounded-full" style="width: ${stepPercentage}%"></div></div>
                        </div>
                    `;
                }).join('');

                pathEarnedXp += phaseEarnedXp;
                const phaseTotalXp = xpTotals.phases[currentPathKey][`phase-${phaseIndex}`];
                const phasePercentage = phaseTotalXp > 0 ? (phaseEarnedXp / phaseTotalXp) * 100 : 0;

                const phaseHtml = `
                    <div>
                        <div class="flex justify-between text-sm mb-1"><span class="font-bold">${phase.Phase.split(':')[0]}</span><span>${phaseEarnedXp} / ${phaseTotalXp} XP</span></div>
                        <div class="w-full bg-slate-200 dark:bg-slate-600 rounded-full h-2.5">
                            <div class="bg-gradient-to-r from-green-400 to-teal-400 h-2.5 rounded-full" style="width: ${phasePercentage}%"></div>
                        </div>
                        <p class="text-xs text-right mt-1 text-slate-500">${phasePercentage.toFixed(0)}% Complete</p>
                        <div class="space-y-3 mt-3">${stepsHtml}</div>
                    </div>
                `;
                activeXpSection.innerHTML += phaseHtml;
                
                if (!allStepsInPhaseComplete) {
                    phaseUnlocked = false; // Lock subsequent phases
                }
            });

            // Add the total path progress bar at the top of the section
            const pathTotalXp = xpTotals.paths[currentPathKey];
            const pathPercentage = pathTotalXp > 0 ? (pathEarnedXp / pathTotalXp) * 100 : 0;
            const totalPathHtml = `
                <div>
                    <div class="flex justify-between text-sm mb-1"><span class="font-bold">${currentPathKey} Progress</span><span>${pathEarnedXp} / ${pathTotalXp} XP</span></div>
                    <div class="w-full bg-slate-200 dark:bg-slate-600 rounded-full h-2.5">
                        <div class="bg-gradient-to-r from-amber-400 to-orange-400 h-2.5 rounded-full" style="width: ${pathPercentage}%"></div>
                    </div>
                    <p class="text-xs text-right mt-1 text-slate-500">${pathPercentage.toFixed(0)}% Complete</p>
                </div>
                <hr class="my-4 border-slate-200 dark:border-slate-700">
            `;
            activeXpSection.insertAdjacentHTML('afterbegin', totalPathHtml);
        },

        // findStepById() - it's just a simple, fast lookup
        findStepById(stepIdToFind) {
            return this.state.stepMap[stepIdToFind] || null;

    // CORRECTED renderCurriculum() - to update XP section on path change
    renderCurriculum(pathKey) {
        this.state.currentPathKey = pathKey; // Keep track of the current path
        const pathData = this.state.curriculumData[pathKey];
        this.state.orderedStepIds = []; // BUG FIX: Reset the ordered list of steps on each render

        if (!pathData || pathData.length === 0) {
            this.nodes.curriculumRoadmap.innerHTML = `<p class="text-slate-500 dark:text-slate-400">No curriculum available for this duration.</p>`;
            return;
        }

        const roadmapHtml = pathData.map((phase, index) => {
            const [phaseLabel, ...phaseTitleParts] = phase.Phase.split(/:\s*/);
            const phaseTitle = phaseTitleParts.join(': ');

            const stepsHtml = phase.Steps.map(step => {
                const stepId = step.Step.replace(/[^a-zA-Z0-9]/g, '-');
                this.state.orderedStepIds.push(stepId); // BUG FIX: Populate the ordered list

                const courses = Array.isArray(step.Courses) ? step.Courses : [step];
                const coursesHtml = courses.map(course => {
                    const altLinkHtml = course['Alternative Free Link']

                            <p><strong><span><i data-feather="shopping-bag" class="w-4 h-4 inline-block -mt-2"></i> Course by: ${course.Provider || 'N/A'}</span></strong></p>

                                <span><i data-feather="star" class="w-4 h-4 inline-block -mt-1"></i> XP: ${course.XPs}</span>

                                <a href="${course['Course Link']}" target="_blank" class="inline-flex items-center gap-1 text-blue-500 hover:underline font-semibold">Go to Course <i data-feather="arrow-right" class="w-4 h-4"></i></a>
                                ${altLinkHtml}

                        <div class="upload-overlay"></div>
                        
                        <div class="absolute top-2 right-2 z-10 flex flex-col items-end gap-2">
                            <button class="mark-complete-btn p-1.5 bg-white/50 dark:bg-slate-800/50 rounded-full hover:bg-green-100 dark:hover:bg-green-900/50 transition">
                                <i data-feather="thumbs-up" class="w-5 h-5 text-green-600 dark:text-white"></i>

                            <h4 class="font-semibold text-lg flex items-center gap-3"><i data-feather="book-open" class="w-5 h-5 text-green-500"></i><span>${step.Step}</span></h4>

                            <div class="dropbox w-[40%] h-48 flex flex-col items-center justify-center rounded-lg bg-black/10 dark:bg-white/5 relative">
                                <input type="file" class="hidden file-input" accept="image/jpeg, image/png">
                                <div class="dropbox-prompt text-center text-white">
                                    <i data-feather="upload-cloud" class="w-10 h-10 mx-auto"></i>
                                    <p class="mt-2 text-sm font-semibold">Drag & Drop or Click</p>

                                </div>

                                    <a href="#" target="_blank" class="view-link inline-flex items-center gap-1 px-3 py-1.5 bg-white text-black text-xs font-bold rounded-full">View</a>

        // After building the main roadmap HTML and putting it on the page:

        this.applyProgressUI();

        // **FIX**: Re-render the XP section for the new path
        this.renderXpSection();
},
    

    /**
     * Updates the UI based on the `userProgress` state.
     * This displays saved images and calls the function to lock/unlock buttons.
     */
    applyProgressUI() {
        document.querySelectorAll('.step-container').forEach(container => {
            const stepId = container.dataset.stepId;
            if (this.state.userProgress[stepId]) {
                const dropbox = container.querySelector('.dropbox');
                this.displayImageInDropbox(this.state.userProgress[stepId], dropbox);
            }
        });
        this.updateCompletionStatus(); // Update lock/complete status
    },

    /**
     * Manages the enabled/disabled/completed state of all "mark as complete" buttons.
     */
    updateCompletionStatus() {
        let firstUncompletedFound = false;
        
        this.state.orderedStepIds.forEach(stepId => {
            const container = document.querySelector(`.step-container[data-step-id="${stepId}"]`);
            if (!container) return;

            const completeBtn = container.querySelector('.mark-complete-btn');
            const isComplete = !!this.state.userProgress[stepId];

            if (isComplete) {
                completeBtn.disabled = false;
                completeBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                completeBtn.innerHTML = `<i data-feather="check-circle" class="w-5 h-5 text-green-600"></i>`;
            } else {
                if (!firstUncompletedFound) {
                    // This is the next step to be completed. Unlock it.
                    completeBtn.disabled = false;
                    completeBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                    completeBtn.innerHTML = `<i data-feather="thumbs-up" class="w-5 h-5 text-green-600 dark:text-white"></i>`;
                    firstUncompletedFound = true;
                } else {
                    // This is a future, locked step.
                    completeBtn.disabled = true;
                    completeBtn.classList.add('opacity-50', 'cursor-not-allowed');
                    completeBtn.innerHTML = `<i data-feather="lock" class="w-5 h-5 text-slate-400"></i>`;
                }
            }
        });
        feather.replace();
    },

    // --- 5. Data Handling (API Calls) ---
    
    /**
     * Fetches progress from the server, then renders the initial curriculum view.
     */
    async fetchAndRenderInitialView() {
        try {
            const response = await fetch(`/api/get_progress/${this.state.goalId}`);
            if (!response.ok) throw new Error('Failed to fetch progress');
            this.state.userProgress = await response.json();
        } catch (error) {
            console.error('Error fetching progress:', error);
            this.state.userProgress = {}; // Ensure progress is an empty object on failure
        } finally {
            // Render the default "Short Duration" path initially
            this.updateSliderUI(0);
            this.renderCurriculum(this.state.pathKeys[0]);
        }
    },

    /**
     * Handles file upload to the backend.
     */

     // CORRECTED handleFile() - to update XP section on new upload
    handleFile(file, dropbox) {
        if (!file || !['image/jpeg', 'image/png'].includes(file.type)) {
            alert('Invalid file type. Please upload a JPG or PNG image.');
            return;
        }

        const stepContainer = dropbox.closest('.step-container');
        const stepId = stepContainer.dataset.stepId;
        
        const formData = new FormData();
        formData.append('file', file);
        formData.append('goal_id', this.state.goalId);
        formData.append('step_id', stepId);

        fetch('/api/upload_proof', {
            method: 'POST',
            body: formData
        })
        .then(response => {
            if (!response.ok) return response.json().then(err => { throw new Error(err.message) });
            return response.json();
        })
        .then(data => {
            if (data.success) {
                this.displayImageInDropbox(data.url, dropbox);
                this.state.userProgress[stepId] = data.url;
                this.updateCompletionStatus();

                // **FIX**: Re-render the XP section with the new progress
                this.renderXpSection(); 
            } else {
                alert('Upload failed: ' + data.message);
            }
        }).catch(error => {
            console.error('Upload fetch error:', error);
            alert('An error occurred during upload.');
        });
    },

    // --- 6. Event Handlers ---

    handleRoadmapClick(e) {
        const button = e.target.closest('.phase-button');
        const completeBtn = e.target.closest('.mark-complete-btn');
        const uploadBtn = e.target.closest('.upload-btn');
        const dropbox = e.target.closest('.dropbox');
        const reuploadLink = e.target.closest('.reupload-link');
        const backBtn = e.target.closest('.back-to-details-btn');

        if (button) {
            button.closest('.phase-container').querySelector('.phase-content').classList.toggle('hidden');
            button.classList.toggle('active');
        } else if (completeBtn) {
            if (completeBtn.disabled) {
                alert("Please complete the previous steps in the roadmap first!");
                return;
            }
            const uploadButton = completeBtn.nextElementSibling;
            uploadButton.classList.toggle('hidden');
            completeBtn.classList.toggle('bg-green-200');
        } else if (uploadBtn) {
            const stepContainer = uploadBtn.closest('.step-container');
            stepContainer.classList.add('upload-view-active');
            stepContainer.querySelector('.back-to-details-btn').classList.remove('hidden');
            uploadBtn.classList.add('hidden');
            stepContainer.querySelector('.step-content-normal').classList.add('hidden');
            stepContainer.querySelector('.step-content-upload').classList.remove('hidden');
        } else if (backBtn) {
            const stepContainer = backBtn.closest('.step-container');
            stepContainer.classList.remove('upload-view-active');
            stepContainer.querySelector('.back-to-details-btn').classList.add('hidden');
            stepContainer.querySelector('.step-content-normal').classList.remove('hidden');
            stepContainer.querySelector('.step-content-upload').classList.add('hidden');
        } else if (reuploadLink) {
            reuploadLink.closest('.dropbox').querySelector('.file-input').click();
        } else if (dropbox && !dropbox.classList.contains('has-file')) {
            dropbox.querySelector('.file-input').click();
        }
    },

    handleFileInputChange(e) {
        if (e.target.matches('.file-input')) {
            this.handleFile(e.target.files[0], e.target.closest('.dropbox'));
        }
    },

    handleDragOver(e) {
        const dropbox = e.target.closest('.dropbox');
        if (dropbox) {
            e.preventDefault();
            dropbox.classList.add('dragover');
        }
    },

    handleDragLeave(e) {
        const dropbox = e.target.closest('.dropbox');
        if (dropbox) {
            e.preventDefault();
            dropbox.classList.remove('dragover');
        }
    },
    
    handleDrop(e) {
        const dropbox = e.target.closest('.dropbox');
        if (dropbox) {
            e.preventDefault();
            dropbox.classList.remove('dragover');
            const file = e.dataTransfer.files[0];
            this.handleFile(file, dropbox);
        }
    },

    // --- 7. UI Helper Functions ---

    displayImageInDropbox(imageUrl, dropbox) {
        dropbox.style.backgroundImage = `url('${imageUrl}')`;
        dropbox.querySelector('.view-link').href = imageUrl;
        dropbox.classList.add('has-file');
    },
    
    updateSliderUI(position, animate = false) {
        const percent = Math.max(0, Math.min(100, position));
        this.nodes.sliderThumb.style.transition = animate ? 'left 0.2s ease' : 'none';
        this.nodes.sliderThumb.style.left = `${percent}%`;

        Object.values(this.nodes.labels).forEach(label => label.classList.remove('active'));
        if (percent < 25) this.nodes.labels.short.classList.add('active');
        else if (percent > 75) this.nodes.labels.long.classList.add('active');
        else this.nodes.labels.moderate.classList.add('active');
    },

    snapSliderAndRender() {
        const currentPos = parseFloat(this.nodes.sliderThumb.style.left);
        let snapPos = 0;
        let pathKey = this.state.pathKeys[0];

        if (currentPos > 75) {
            snapPos = 100;
            pathKey = this.state.pathKeys[2];
        } else if (currentPos > 25) {
            snapPos = 50;
            pathKey = this.state.pathKeys[1];
        }
        
        this.updateSliderUI(snapPos, true);
        this.renderCurriculum(pathKey);
    }
};

    // --- Start the application once the DOM is ready ---
    document.addEventListener('DOMContentLoaded', () => {
        CurriculumManager.init();
    });

</script>

    // --- New Slider Functions ---
    function updateSliderUI(position, animate = false) {
        let percent = Math.max(0, Math.min(100, position));
        sliderThumb.style.transition = animate ? 'left 0.2s ease' : 'none';
        sliderThumb.style.left = `${percent}%`;

        Object.values(labels).forEach(label => label.classList.remove('active'));
        if (percent < 25) labels.short.classList.add('active');
        else if (percent > 75) labels.long.classList.add('active');
        else labels.moderate.classList.add('active');
    }

    function snapSliderAndRender() {
        const currentPos = parseFloat(sliderThumb.style.left);
        let snapPos = 0;
        let pathKey = pathKeys[0];

        if (currentPos > 75) {
            snapPos = 100;
            pathKey = pathKeys[2];
        } else if (currentPos > 25) {
            snapPos = 50;
            pathKey = pathKeys[1];
        }
        
        updateSliderUI(snapPos, true);
        renderCurriculum(pathKey);
    }

    sliderThumb.addEventListener('mousedown', () => { isDragging = true; });
    document.addEventListener('mouseup', () => {
        if (isDragging) {
            isDragging = false;
            snapSliderAndRender();
        }
    });
    document.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        const rect = sliderContainer.getBoundingClientRect();
        let newX = e.clientX - rect.left;
        let percent = (newX / rect.width) * 100;
        updateSliderUI(percent);
    });
    

    // Allow clicking on labels to move the slider
    Object.entries(labels).forEach(([key, label], index) => {
        label.addEventListener('click', () => {
            const snapPos = index * 50;
            updateSliderUI(snapPos, true);
            renderCurriculum(pathKeys[index]);
        });
    });
    
// --- curriculumRoadmap event listener ---
curriculumRoadmap.addEventListener('click', (e) => {
    const button = e.target.closest('.phase-button');
    const completeBtn = e.target.closest('.mark-complete-btn');
    const uploadBtn = e.target.closest('.upload-btn');
    const dropbox = e.target.closest('.dropbox');
    const reuploadLink = e.target.closest('.reupload-link');
    const backBtn = e.target.closest('.back-to-details-btn');

    // Handle Phase Button Clicks
    if (button) {
        const targetContainer = button.closest('.phase-container');
        const targetContent = targetContainer.querySelector('.phase-content');
        button.classList.toggle('active');
        targetContent.classList.toggle('hidden');
    }

    // Handle "Mark as Complete" Button
    if (completeBtn) {
        const uploadButton = completeBtn.nextElementSibling;
        uploadButton.classList.toggle('hidden');
        completeBtn.classList.toggle('bg-green-200');
    }

    // Handle "Upload" Button
    if (uploadBtn) {
        const stepContainer = uploadBtn.closest('.step-container');
        stepContainer.classList.add('upload-view-active');
        stepContainer.querySelector('.back-to-details-btn').classList.remove('hidden');
        stepContainer.querySelector('.upload-btn').classList.add('hidden');
        stepContainer.querySelector('.step-content-normal').classList.add('hidden');
        stepContainer.querySelector('.step-content-upload').classList.remove('hidden');
    }

    // Handle "Back to Details" Button
    if (backBtn) {
        const stepContainer = backBtn.closest('.step-container');
        stepContainer.classList.remove('upload-view-active');
        stepContainer.querySelector('.back-to-details-btn').classList.add('hidden');
        stepContainer.querySelector('.step-content-normal').classList.remove('hidden');
        stepContainer.querySelector('.step-content-upload').classList.add('hidden');
    }
    
    // Handle Dropbox Click (to open file dialog)
    if (dropbox && !dropbox.classList.contains('has-file')) {
        dropbox.querySelector('.file-input').click();
    }
    
    // Handle Re-upload Click
    if (reuploadLink) {
        reuploadLink.closest('.dropbox').querySelector('.file-input').click();
    }
});

        };
